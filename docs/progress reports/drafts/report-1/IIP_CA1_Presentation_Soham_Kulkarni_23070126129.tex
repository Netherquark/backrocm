\begin{frame}{Introduction}
\begin{itemize}
        \item Conducted as an employee of \textit{FOSS United Foundation}.
        \item ROCm is AMD's CUDA
        \item Primary target: \texttt{gfx90c} (Ryzen 5 5650U); validation target: \texttt{gfx902} (Ryzen 5 2200G).

\begin{frame}{Problem statement}
\begin{itemize}
        \item Native HIP execution succeeds, but packaged ROCm libraries lack complete gfx90c support.
        \item Missing rocBLAS/Tensile artifacts prevent higher-level workloads such as PyTorch.
        \item Community spoofing enables evaluation but is not a production-grade solution.
        \item A reproducible recovery workflow for unsupported architectures is currently unavailable.
\end{itemize}
\end{frame}


\begin{frame}{Objectives of the project}
\begin{itemize}
        \item Recover functional ROCm support for gfx90c using reproducible methods.
        \item Benchmark every recovery stage against the native software stack \& against spoofing.
        \item Validate the methodology on gfx902.
        \item Generalise the workflow for other unsupported AMD architectures.
    \end{itemize}
\end{frame}

    \end{itemize}

\begin{frame}{Gap in the research, technology, and methodology}
\vspace{-0.5cm}
\begin{itemize}
        \item No documented end-to-end workflow for recovering unsupported ROCm targets.
        \item Packaged ROCm components omitted \texttt{gfx90c}-specific code generation and runtime data.
        \item Recovery knowledge was fragmented across documentation, issue trackers, pull requests, and community discussions.
        \item No methodology existed to validate recovery, benchmark correctness, and reproduce results across GPU generations.
    \end{itemize}
\end{frame}

\begin{frame}{Description of the proposed solution}
\begin{itemize}
        \item Establish a staged recovery workflow from native binaries to complete source builds.
        \item Validate every layer using HIP applications, \texttt{rocm-examples}, and PyTorch workloads.
        \item Isolate compiler, runtime, library, packaging, and hardware-specific failures independently.
        \item Produce a reproducible workflow transferable to other unsupported AMD GCN GPUs.
\end{itemize}
\end{frame}

\begin{frame}{Native ROCm baseline}
\vspace{-0.5cm}
\begin{itemize}
    \item Installed ROCm 6.4.4 on Fedora 43 for \texttt{gfx90c}.
    \item Verified HIP compilation and execution using native ROCm binaries.
    \item Identified PyTorch failures caused by missing \texttt{gfx90c} rocBLAS Tensile data.
    \item Evaluated HSA architecture spoofing as an initial compatibility baseline.
    \item Ran Fedora provided \texttt{rocm-examples} as a baseline.
\end{itemize}
\end{frame}


\begin{frame}{Source recovery}
\vspace{-0.5cm}
\begin{itemize}
    \item Built \texttt{rocm-examples} from source using the ROCm 6.4.4 toolchain.
    \item Investigated missing \texttt{main\_gfx90c.s} assembly generation.
    \item Extended the \texttt{assembly\_to\_executable} workflow for \texttt{gfx90c}.
    \item Traced compilation from HIP source through LLVM IR to AMDGPU assembly.
\end{itemize}
\end{frame}


\begin{frame}{Failure analysis}
\vspace{-0.5cm}
\begin{itemize}
    \item Classified failures into packaging, metadata, missing Tensile data, and hardware limitations.
    \item Compared behaviour across native, spoofed, and source-built environments.
    \item Validated each recovery stage using HIP applications and \texttt{rocm-examples}.
    \item Recorded reproducible failure cases for subsequent recovery work.
\end{itemize}
\end{frame}


\begin{frame}{Benchmark evaluation}
\vspace{-0.5cm}
\begin{itemize}
    \item Evaluated memory bandwidth using CLBlast and clpeak.
    \item Measured matrix multiplication performance using rocBLAS benchmark clients.
    \item Compared application behaviour under native and spoofed execution.
    \item Used benchmark results to validate recovered functionality and identify remaining regressions.
\end{itemize}
\end{frame}

\begin{frame}{Intended project outcome}
\vspace{-0.5cm}
\begin{itemize}
    \item A reproducible workflow for recovering unsupported AMD GCN GPUs under ROCm.
    \item Complete \texttt{gfx90c} enablement across the ROCm software stack.
    \item Validation of the recovery methodology on \texttt{gfx902}.
    \item Benchmark-backed evaluation of native, spoofed, and recovered software stacks.
    \item Documentation suitable for upstream contribution and long-term maintenance.
\end{itemize}
\end{frame}


\begin{frame}{Conclusion}
\vspace{-0.5cm}
\begin{itemize}
    \item Native ROCm functionality on unsupported hardware was successfully characterised.
    \item Compiler, runtime, packaging, and library limitations were systematically isolated.
    \item A staged recovery methodology was established for unsupported ROCm targets.
    \item The proposed workflow provides a foundation for future upstream enablement of legacy AMD GPUs.
\end{itemize}
\end{frame}
